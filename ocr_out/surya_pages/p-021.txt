قال سليمان الجمزوري (رحمه الله) :
أولاهُمَــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
للأم اَلْ حَـالان قَـبْـــلَ الأَحْــرُف
من ابّسغ حَجُّكَ وَخَفَتْ عَقَيْمَـمَـهُ
و انانالاصحات
قبل أربع مَعْ عشرة خُذٌ علّمَـــهُ
ثانيــهــمَـــا إدغــامُــهَــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
وعَشْرَة أَيضَاً وَرَمْـْزَهَــا فـــع
طبّ تُمّ صلّ رَحمَّا تَفُرْ ضَفَ ذَا نَعَمْ
دَ عٌ سُسُوءَ ظَــن زُرٌ شَريفًا للّكَـرَمْ
واللاَّمَ الأُخْرَى سَمَّبْهَا شَمْسَيَّة
واللامَ الأولى سَمَّمُّهَا قَمْتْريّـــة
ويجو دالدوار
الأسئلة
١. اذكر أنــواع اللامـــات الســــواكن . وكم حالة للام ال قبل أحرف
الهجاء ؟
٢. كم حـرفًا تختـص باللام القمـــريــة ؟ وما حكمهـــا ؟ وما وجـــه
تسميتـــه إظهاراً قـمريًا ؟ وما سبب الإظهار ؟ مثل لكل حرف
بثال .
٣. كم حرفا تختص باللام الشمسية ؟ ومـا حكمـهـا عند هذه
الأحـرف؟ وما وجــه تسميته إدغامًا شمسيًا ؟ وما سببـه؟ مثل
لكل حرف بمثال ؟
