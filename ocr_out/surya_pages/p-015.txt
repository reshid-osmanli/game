حــذر أهــل الأداء من إخــفــاء الميم الســـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
لاتحاد مخرجها مع الواو، وقرب مخرجها من الفاء، وذلك لأن حكمها
الإظهار عندهما ، مـثــال الميم عند الفـاء : ﴿ذَرْأَكُمْ فِى ٱلْأَرْضِ ﴾١١ ،
تائمال صحات
ومثال الميم عند الواو : ﴿عَلَيَهـمْرُوُلَا أَلَضَتَآلَينَ ﴾(٢)(*)
قال سليمان الجمزورى (رحمه الله) :
واحذر لدا واو وفَا أن تختَفي لقَرْبَهُـــــا والاتَّحَــــاد فاعــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
كيفية الأداء عند النطق بالميم الساكنة قبل حرفي الواو والفاء :
يكون الأداء بتقوية الاعتماد على مخرجها أي على الشفتين،
وبإظهار سكونها بلا إظهار غنة، فزمان إظهار الميم لعدم ظهور
الغنة اسرع من زمان اخفائها (۳) .
الأمور التي ينبغي على القارئ أن يحترز منها عند آدائه لإظهار
الملام الساكنة :
– يحترز القارئ من السكت على الميم الساكنة.
– يحترز القارئ من الإتيان بالقلقلة فى الميم الساكنة.
- يحترز القارئ من الإتيان بغنة زائدة عن الصفة الذاتيــة لحرف
الميم.
( ١ ) سورة الملك الأية رقم ( ٢٤ ).
(٢) سورة الفاتحة الاية رقم (٧).
(٣) نهاية القول المفيد ص ١٢٨ .
( *) لا تدخم الميم في الواو وإن تجانسا في المخرج، فرقاً بينها وبين النون المدغمة في الواو، وخوفاً من اللبس.
