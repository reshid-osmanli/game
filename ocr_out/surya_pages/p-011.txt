قال الإمام ابن الجزري (رحمه الله)
مَــت الزَّخْـرُف بالتَّــــرُف بالتَّــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
اع انتالا صحات
الاعراف روم هــود كـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
نعـــمـــتُـــهَـــــــُـــــهَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ

- عَلى تجويد الكتاب
ه سُفَ عــمـــ اَنَ الْقَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
