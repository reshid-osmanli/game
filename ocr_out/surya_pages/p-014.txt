بينهما مخرج فهما متقاربان ، وإن كانا من عضو واحد فهما
متقاربان إن لم يوجـد مخـرج فاصـل بينهما كاقصـى الحلـق مع
وسطه، وإلا فمتباعدان كأقصاه مع أدناه، وقد يكون التقارب فى
يو اندالاصحات
الصفات دون المخارج، والله أعلم .
قال سليمان الجمزورى (رحمه الله) :
إنْ في الصَّـفَات والمخــارج اتَّقَقٌ حـرفــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
وإن يَكـــونَا مَـخْرَجــًا تَقَارَبَــا وفى الصَّـفَـات الْحْتَـلْفَـا يُلَقَّبَا
- عالى يجويدالكتاب
مُتَقَارَبَـيْنَ أوْ يَكُــونَــا اتٌفَقَــا    في مَـخْرَجِ دُونَ الصَّفات حُقْقَا
بالْمُتَجَــانسَـيْنِ شَـمٌ إِن سَـــكَنٌ اأوُلٌ كُــلٌ فـالصـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
أو حُركَ الْحَرْفَان في كُلَّ فَقُل كَــلٌ كَـــلٌ كَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ

وَآوَآلَـــى مــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
