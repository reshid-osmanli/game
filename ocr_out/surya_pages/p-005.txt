قال الإمام الشاطبي (رحمه الله) :
وكُلُهُــمُ التنويــنَ والنــــونَ ادغَمُـــــــوا
بلا غَنَّت في الّــــلام والــراً ؛ لَيَجْمُـــلأ
اع اندالا صحات -
وكُـلٌ بيَنْمُــــو ادْعَمُـــــوا مَــــعَ غُنَّـــــةٍ 
وفي الْـــوَاو والْيَـــا دُونَهَــا خَلَــفَ تَــلاَ
وَعنْدهُـمَــــا للَكُـــلٌ اظْهـــرٌ بكلمَـــــةٍ 
مَخَافَـةَ إِشْبَـــاه الْمُضَـــاعَــــف أثْقــــلأ
وعندَ حُـرُوف الحْلُــــﻖ للكُـــلَّ أظَّـهــــراَ
ألا هَـاجَ حُكُمُ عَــمٌ خَاليــــه عُفّــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
وقلبهُمَا ميمًا لـــدى البَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
عَلَـــى غُنَّة عنـــدَ الْبَواقـــي ليَكُمُـــلأ
قال الإمام ابن الجزري (رحمه الله) :
وحكــــــم تنويـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
إظهــــــارٌ ادْغـــــــــامٌ وقلـــبٌ إخْفَــــــــا
فعندً حــرف الْحَلّـق أظهــــرْ وادَّغـــمْ
في الــلاّم والــــرّاً لا بغنــــــة لَــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
وأدْعُمـــــن بغنـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
إلا بكلمـــــة كدنــــيــــــا عَنْــوَنُــــــــــــا عَنْــوَنُــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
والقَلْـــبَ عنــد البـــَا بغنــــــة كــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
الاخْفَـا لـــدى باقــــي الحُــــــي الحـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
(١) المراد أن حروف الإظهار هى أوائل هذه الكلمات .
وقال أبو شامة (رحمه الله) فى كتابه (إبراز المعانى) ص ٢٠٢ باب أحكام النون الساكنة،
والتنوين قوله خَاليه : أى ماضيه .
غفلاً : جمع غافل ـ وكأنه أشار بهذا الكلام إلى الموت، أو إلى البعث الذي عمّ الغافلين عنه .
(٢) إذا أضيف الاسم المفرد لما يعده فإنه قد يراد به الجمع، نحو : ﴿واذكروا نعمة اللّه﴾ .
