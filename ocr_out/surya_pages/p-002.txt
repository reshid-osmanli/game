ح. الراء المكسسورة التى سكنت لأجــل الوقــف، وقــد سـبـقــهـا
فتــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
﴿ بِشَـرَرِ ﴾(١) فَــالراء الأولى مــفـخــمـة، وهذا يستــدعى
تانانة اللهوات
تفخيم الثانية الساكنة لأجل الوقف .
ط. الراء المفتوحة، او المضمومة التى سكنت لأجل الوقف، وقد
سبقها ساكن مسبوق بفتح، أو ضم، نحو: هاإنَّ ٱلْأَمْرَ﴾(٢١،
هاآلَيْسَتَـرَ (٣))، هُنصَّــَ ُ ﴾ ١)، هِ حَضَرُ ﴾(٥) وكذا إذا
كان الساكن ألفًا، أو واوً، نحو: ۞ألثّارَ بُحو : ﴿أَلّْأَمْمُ ‫( ١١)،
عاليجوندالكتار
ويستثني اليا ء، نحو: ﴿ ألسَّّيَّ بُ (٨) فالراء مرققة؛ لوجود
الناء الساكنة قبلها .
قال الإمام ابن الجزري (رحمه الله) :
ورُقّــتّ الــــرَأَ ءَ إذا مـــا كُــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
كـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
إن لم تـگن من قــبل حــرف اســتــعُـلا
أو كــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
والحُلّـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
وأخـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
