صفات الصروف
الصفــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
لغــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
سكانةالأصحيات
فالمقصود هنا: المعانى الحسية، او المعنوية، وليس معنى
النعت كما فى النحو .
واصطلاحاً : كيفية ثابتة للحرف عند النطق به من جهر، واستعلاء،
وقلقلة، ونحو ذلك .
JESSI JUSESC
فوائل الصفات :
١. تمييز الحروف المشتركة فى المخرج .
٢. معرفة القوي من الضعيف؛ ليعلم ما يجوز ان يدغم، وما لايجوز ،
فان ماله قوة ومزية على غيره لا يجوز ان يدغم فى ذلك الغير؛
لئلا تذهب تلك المزية (والعبرة فى ذلك بالرواية) .
٣. تحسين لفظ الحروف المختلفة المختلفة المخارج .
عدد الصغارت :
اختلف العلماء في عدد الصفات: فمنهم من عدّها سبع عشرة
صفة، وهو الإمـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
شراح مقدمته كالمًالاً على وغيره، ومنهم من زاد على ذلك إلى ان
أوصلها اربعاً واربعين صفة ، ومنهم من نقص عـن السبع عشـرة،
وعــدها أربــع عــشــرة بنقص الذلاقــة، والإصــمــات، والانحــراف،
واللين ، وزيادة صفة الغنة ـ وغير ذلك من الحلاف فى عدها .
