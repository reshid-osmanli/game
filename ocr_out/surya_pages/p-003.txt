وقد أشار إلى بعض ذلك الإمام السخاوي في منظومته بقوله:
لا تحــسَب التّــجْــويدَ مَــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
أو مـــدَّ مــــا لا مَــــدَّ فــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
ان اندالا صحاب الأماك الأماك الأمر الأندار الأندار الأندار الأندار الأند
أو أنْ تُشَــــدِّدَ بَعْـــــدَ مَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
أو أنّ تَلُوكَ الحّـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
أو أنَّ تَفُـــوهَ بـهَـــمْـــزَةٍ مُـــتَـــهَـــوٌ عَــــا
- عَلىتجويدالكِكاب
فَــيَــفَــرٌ سَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
للْخَـرْفِ مـــيـــزانٌ فَــلا تَكُ طَاغـــيًـــا
فـــيـــه ولا تَكَ مُــــخــــــــرَ الميـــــزَانِ 
فــاذا هَمَـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
من غَــيْــر مَــا بُهْــرِ (٢) وغَـــيْـــر تَوَان
وامْــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
أو هُمْـــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
فكل حرف له ميزان يعرف به مقدار حقيقته، وذلك الميزان هو
مخرجه وصفته فإذا خرج من مخرجه معطى ما له من الصفات على
وجه العدل في ذلك من غير إفراط ولا تفريط فقد وُزن بميزانه، وهذا
هو حقيقة التجويد (۳) .
(١) أى المتأنى.
(٢ )البُهّرُ : تتابع النَّفَس من الإعياء والجهد.
(٣) نهاية القول المفيد (بتصرف) ص ٢٠ ، ٢١ .
